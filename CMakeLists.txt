cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
    set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "Supported macOS architecture")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(ITUPickerProject VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# The generator expression prevents an extra Release/ subdirectory on MSVC.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")

if(APPLE)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64"
       OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang" OR CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "14.0")
        message(FATAL_ERROR "macOS requires Apple Silicon, Apple Clang and deployment target 14+")
    endif()
    set(ITU_TARGET macos-arm64)
elseif(WIN32)
    if(NOT MSVC OR CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MSVC_VERSION LESS 1930 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8
       OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
        message(FATAL_ERROR "Windows requires Visual Studio 2022 MSVC with -A x64 (use windows-x64 preset)")
    endif()
    set(ITU_TARGET windows-x64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU"
       OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS "11")
        message(FATAL_ERROR "Linux requires x64 and GCC 11+")
    endif()
    set(ITU_TARGET linux-x64)
else()
    message(FATAL_ERROR "Supported targets: Windows x64, macOS ARM64, Linux x64")
endif()
if(MSVC)
    add_compile_options(/W4 /utf-8)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
include(cmake/Dependencies.cmake)

if(WIN32)
    set(ITU_PLATFORM_SOURCE src/platform_windows.cpp)
    set(ITU_TEST_SOCKET_SOURCE src/http_socket_windows.cpp)
else()
    set(ITU_PLATFORM_SOURCE src/platform_posix.cpp)
    set(ITU_TEST_SOCKET_SOURCE src/http_socket_posix.cpp)
endif()
add_library(itu_platform STATIC ${ITU_PLATFORM_SOURCE})
target_include_directories(itu_platform PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
    target_link_libraries(itu_platform PUBLIC advapi32 shell32 winmm)
endif()
set(ITU_CORE_SOURCES src/http.cpp src/token.cpp src/clock.cpp)
add_library(itu_core STATIC ${ITU_CORE_SOURCES})
target_include_directories(itu_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(itu_core PUBLIC CURL::libcurl itu_platform)
add_executable(main src/main.cpp)
target_link_libraries(main PRIVATE itu_core)
add_executable(setup setup/main.cpp)
target_link_libraries(setup PRIVATE itu_platform)

# A machine-readable package input; no absolute paths or user environment values.
execute_process(COMMAND git log -1 --format=%H WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE ITU_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT ITU_REVISION MATCHES "^[0-9a-f]+$")
    set(ITU_REVISION unknown)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/bin/build-manifest.json" CONTENT
"{\n  \"schema_version\": 1,\n  \"version\": \"${PROJECT_VERSION}\",\n  \"target\": \"${ITU_TARGET}\",\n  \"compiler\": {\"id\": \"${CMAKE_CXX_COMPILER_ID}\", \"version\": \"${CMAKE_CXX_COMPILER_VERSION}\"},\n  \"revision\": \"${ITU_REVISION}\",\n  \"dependencies\": ${ITU_DEPENDENCY_MANIFEST}\n}\n")

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
    add_library(itu_core_test STATIC ${ITU_CORE_SOURCES} ${ITU_TEST_SOCKET_SOURCE})
    target_include_directories(itu_core_test PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(itu_core_test PUBLIC CURL::libcurl itu_platform)
    target_compile_definitions(itu_core_test PUBLIC ITU_ENABLE_TEST_SEAMS)
    if(WIN32)
        target_link_libraries(itu_core_test PRIVATE ws2_32)
    endif()
    add_executable(transport_test tests/transport_test.cpp)
    target_link_libraries(transport_test PRIVATE itu_core_test)
    add_test(NAME transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/transport_fixture.py" "$<TARGET_FILE:transport_test>")
    add_executable(auth_driver tests/auth_driver.cpp)
    target_link_libraries(auth_driver PRIVATE itu_core_test)
    add_test(NAME authentication COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/auth_fixture.py" "$<TARGET_FILE:auth_driver>")
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE itu_core)
    add_test(NAME core_and_clock COMMAND core_tests)
    add_executable(application_driver tests/application_driver.cpp)
    target_link_libraries(application_driver PRIVATE itu_core_test)
    add_test(NAME application_flags COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/application_fixture.py" "$<TARGET_FILE:application_driver>")
    add_executable(setup_file_tests tests/setup_file_tests.cpp)
    target_link_libraries(setup_file_tests PRIVATE itu_platform)
    if(MSVC)
        target_compile_options(setup_file_tests PRIVATE /UNDEBUG)
        add_executable(windows_console_tests tests/windows_console_tests.cpp)
        target_link_libraries(windows_console_tests PRIVATE itu_platform)
        add_test(NAME setup_terminal COMMAND windows_console_tests "$<TARGET_FILE:setup>")
    else()
        target_compile_options(setup_file_tests PRIVATE -UNDEBUG)
        add_test(NAME setup_terminal COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/setup_pty_tests.py" "$<TARGET_FILE:setup>")
    endif()
    add_test(NAME setup_files COMMAND setup_file_tests)
    set_tests_properties(transport authentication core_and_clock application_flags setup_files setup_terminal PROPERTIES TIMEOUT 90)
endif()
